# 親のテキストボックスを子で入力させる

## 親のテキストボックスに値を更新する関数を子に props で渡す

### 親

```
"use client";
import Child from "../_components/Child";
import styles from "../page.module.css";
import { useState } from "react";
export default function ParentPage() {
  const [text, setText] = useState<string>("");
  const updText = (str: string) => setText(str);
  console.log('text', text);
  return (
    <div className={styles.page}>
      <Child updText={updText} />
    </div>
  );
}
```

### 子

```
import { TextField } from '@mui/material';
type Props = {
  updText: (str: string) => void
}
export default function Child({updText}:Props) {
  return (
    <TextField
      onChange={(e) => updText(e.target.value)}
    />
  )
}
```
